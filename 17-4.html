核心代码

async function sc_sendRequest() {
    const sc_options = {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ title: '沈畅的接口测试', userId: 1 })
    };

    const sc_res = await fetch('https://jsonplaceholder.typicode.com/posts', sc_options);

    // 4xx / 5xx 不会抛错，需要自己判断
    if (!sc_res.ok) {
        throw new Error('HTTP ' + sc_res.status + ' ' + sc_res.statusText);
    }

    // 2xx 才继续解析响应体
    const sc_data = await sc_res.json();
    console.log(sc_data);
}

注意：fetch 只在网络层失败时 reject，HTTP 404 / 500 属于「请求成功但有错误状态」，必须用 res.ok 或 res.status 自行判断。
