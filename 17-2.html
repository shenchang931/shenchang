fetch 获取用户列表

使用原生 fetch() 请求公开接口 http://jsonplaceholder.typicode.com/users。 该接口允许跨域访问，返回一个包含 10 位用户的 JSON 数组， 每个用户对象里有 name、username、email、phone、 website，以及嵌套的 address（含 geo 经纬度）和 company 对象。

核心代码

async function sc_loadUsers() {
    // 1. 发起请求，拿到响应对象
    const res = await fetch('http://jsonplaceholder.typicode.com/users');

    // 2. 判断 HTTP 状态码，非 200 直接抛错
    if (!res.ok) throw new Error('HTTP 状态码 ' + res.status);

    // 3. 响应体解析为数组，共 10 个用户对象
    const data = await res.json();

    // 4. 遍历数组渲染卡片，嵌套字段用链式访问
    data.forEach(function (user) {
        console.log(user.name, user.address.city, user.company.name);
    });
}
